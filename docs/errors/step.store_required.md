# `step.store_required`

An action makes several changing calls but writes no database, and the app names no database to keep the record of its calls in.

## Why Jadpo refuses it

A changing call, such as a provider operation with `changes: true` or a
store's `put`, takes effect as soon as it is made. When an action makes
several, Jadpo records each one before it goes out and its answer when it
comes, so a retry replays the answered calls instead of sending them
again. That record is kept in a database, beside the outbox of the one
the action writes. An action that writes none has nowhere to keep it,
and Jadpo won't choose a database for it: `events: { store: }` names
the one.

## What it looks like

domains/bakery/actions/file_prices.jadpo, in a bakery app whose only
connection is `sheets`, a folder of files:

```jadpo check project=e-step-store-required file=app.jadpo hidden
application {
    connections: {
        sheets: {
            driver: Driver.disk
            path: "/var/lib/bakery/sheets"
        }
    }
    steps: {
        encryption_key: config.step_encryption_key
    }
}
```

```jadpo check project=e-step-store-required file=config.jadpo hidden
config {
    step_encryption_key: SigningKey
}
```

```jadpo check project=e-step-store-required file=domains/bakery/types/price_list.jadpo hidden
/// A supplier's prices, one line for each loaf.
type PriceList = Text {
    min_length: 1
    max_length: 2000
}
```

```jadpo check project=e-step-store-required file=domains/bakery/failures/sheet_not_kept.jadpo hidden
/// The sheets store had no room for a sheet.
type SheetNotKept = Failure {
    kind: FailureKind.unavailable
    message: "There was no room to keep the sheet."
}
```

```jadpo check project=e-step-store-required file=domains/bakery/events/prices_changed.jadpo hidden
use bakery/types { PriceList }

/// A supplier sent new prices.
type PricesChanged = Event {
    price_list: PriceList
}
```

```jadpo check project=e-step-store-required file=domains/bakery/events/events.jadpo_record hidden

record events 1

event PricesChanged
    price_list: Text { min_length: 1, max_length: 2000 }
```

```jadpo check project=e-step-store-required file=domains/bakery/subscribers/file_new_prices.jadpo hidden
use bakery/actions { file_prices }
use bakery/events { PricesChanged }

/// Files each supplier's new prices.
subscriber file_new_prices {
    on: PricesChanged
    run: file_prices(event)
}
```

```jadpo check project=e-step-store-required file=domains/bakery/actions/file_prices.jadpo expect=step.store_required
use app { sheets }
use bakery/failures { SheetNotKept }
use bakery/events { PricesChanged }

/// Keeps a supplier's new prices as a sheet, then a copy for the
/// accountant.
action file_prices(event: PricesChanged) fails SheetNotKept -> Unit {
    var sheet: File = attempt sheets.put(encoding.to_utf8(event.price_list), FileType.csv) {
        storage_full: SheetNotKept
    }
    var invoice_copy: File = attempt sheets.put(encoding.to_utf8(event.price_list), FileType.csv) {
        storage_full: SheetNotKept
    }
    assert sheet != invoice_copy
}
```

`(human)` at the end of the `next:` line tells your coding agent to stop
and ask you which fix you want, because the choice is yours to make: it
decides which database keeps the record of these calls, and the events
of every action that writes none.

## How to fix it

**If a database should keep the record**, connect it and name it in
`events: { store: }` in `app.jadpo`, with its address as a setting in
`config.jadpo`. Here the database is a new one, `main`:

```jadpo fragment
        main: {
            driver: Driver.postgres
            url: config.main_url
        }
    events: {
        store: main
    }
```

```jadpo check project=e-step-store-required-fix-1 file=app.jadpo hidden
application {
    connections: {
        main: {
            driver: Driver.postgres
            url: config.main_url
        }
        sheets: {
            driver: Driver.disk
            path: "/var/lib/bakery/sheets"
        }
    }
    events: {
        store: main
    }
    steps: {
        encryption_key: config.step_encryption_key
    }
}
```

```jadpo check project=e-step-store-required-fix-1 file=config.jadpo hidden
config {
    main_url: PostgresUrl
    step_encryption_key: SigningKey
}
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/types/price_list.jadpo hidden
/// A supplier's prices, one line for each loaf.
type PriceList = Text {
    min_length: 1
    max_length: 2000
}
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/failures/sheet_not_kept.jadpo hidden
/// The sheets store had no room for a sheet.
type SheetNotKept = Failure {
    kind: FailureKind.unavailable
    message: "There was no room to keep the sheet."
}
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/events/prices_changed.jadpo hidden
use bakery/types { PriceList }

/// A supplier sent new prices.

    price_list: PriceList
}
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/events/events.jadpo_record hidden

    price_list: Text { min_length: 1, max_length: 2000 }
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/subscribers/file_new_prices.jadpo hidden
use bakery/actions { file_prices }
use bakery/events { PricesChanged }

/// Files each supplier's new prices.
subscriber file_new_prices {
    on: PricesChanged
    run: file_prices(event)
}
```

```jadpo check project=e-step-store-required-fix-1 file=domains/bakery/actions/file_prices.jadpo hidden migrate=initial
use app { sheets }
use bakery/failures { SheetNotKept }
use bakery/events { PricesChanged }

/// Keeps a supplier's new prices as a sheet, then a copy for the
/// accountant.
action file_prices(event: PricesChanged) fails SheetNotKept -> Unit {
    var sheet: File = attempt sheets.put(encoding.to_utf8(event.price_list), FileType.csv) {
        storage_full: SheetNotKept
    }
    var invoice_copy: File = attempt sheets.put(encoding.to_utf8(event.price_list), FileType.csv) {
        storage_full: SheetNotKept
    }
    assert sheet != invoice_copy
}
```

Each of `file_prices`'s two `sheets.put` calls is now recorded in `main`
before it goes out, and its answer after it. When the delivery is tried
again, the answered calls are replayed rather than sent again, and the
record is deleted when the action commits.

An action that writes a database keeps its record in that one instead,
so it never needs `events: { store: }`.
